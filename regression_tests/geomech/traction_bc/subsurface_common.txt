#=========================== numerical methods ================================
NUMERICAL_METHODS FLOW

  TIMESTEPPER
    MAX_TS_CUTS 200
  /

  NEWTON_SOLVER
    RTOL 1.000d-08
    STOL 1.000d-30
    ITOL 1.000d-08
    ITOL_UPDATE 0.0001
  /
END

EOS WATER
  DENSITY CONSTANT 1.0 #[kg/m^3]
  VISCOSITY CONSTANT 1.0 #[Pa-s]
END

GRID
  TYPE STRUCTURED
  BOUNDS
    0.0 0.0 0.0 
    1.0 1.0 1.0 
  /
  NXYZ 1 1 1
  GRAVITY 0.0 0.0 0.0 
END

TIME
  FINAL_TIME 1.d0 s
  INITIAL_TIMESTEP_SIZE 1.d0 s
  MAXIMUM_TIMESTEP_SIZE 1.d0 s
END

MATERIAL_PROPERTY soil1
  ID 1
  CHARACTERISTIC_CURVES default
  POROSITY 0.3
  TORTUOSITY 1.0
  ROCK_DENSITY 1.d0
  PERMEABILITY
    PERM_ISO 1.000d-4
  /
  GEOMECHANICS_SUBSURFACE_PROPS
    COMPRESSIBILITY_FUNCTION LINEAR_MODEL
  /
END

skip
OUTPUT
  PERIODIC TIMESTEP 1
  FORMAT HDF5
  VARIABLES 
    POROSITY
    LIQUID_PRESSURE
    LIQUID_SATURATION
    LIQUID_DENSITY
    PERMEABILITY
  /
END
noskip

  SATURATION_FUNCTION VAN_GENUCHTEN

   M 0.5
   LIQUID_RESIDUAL_SATURATION 0.1
  / 
  PERMEABILITY_FUNCTION MUALEM_VG_LIQ
   M 0.5
   LIQUID_RESIDUAL_SATURATION 0.1
  / 
END

    1.0 1.0 1.0 
  /
END

    0.0 0.0 0.0

END
